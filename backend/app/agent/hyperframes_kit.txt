// ---- kit helpers: every function adds tweens to `tl` at absolute seconds ----
// Never animate a .clip element itself; animate elements INSIDE each scene.
function fadeUp(tl, sel, at, dur, dy){ // rise + fade in; dy defaults to 24px
  tl.fromTo(sel, {opacity:0, y:(dy==null?24:dy)},
    {opacity:1, y:0, duration:(dur==null?0.6:dur), ease:"power3.out"}, at||0);
}
function fadeIn(tl, sel, at, dur){ // plain fade in
  tl.fromTo(sel, {opacity:0}, {opacity:1, duration:(dur==null?0.5:dur), ease:"power2.out"}, at||0);
}
function pop(tl, sel, at, dur, s){ // small scale-in, good for badges/numbers
  tl.fromTo(sel, {opacity:0, scale:(s==null?0.92:s), transformOrigin:"50% 50%"},
    {opacity:1, scale:1, duration:(dur==null?0.5:dur), ease:"back.out(1.6)"}, at||0);
}
function fadeOut(tl, sel, at, dur){ // fade an element out mid-scene
  tl.to(sel, {opacity:0, duration:(dur==null?0.4:dur), ease:"power2.in"}, at||0);
}
function drawPath(tl, sel, at, dur){ // SVG stroke draw-on; requires pathLength="1"
  tl.fromTo(sel, {strokeDasharray:1, strokeDashoffset:1},
    {strokeDashoffset:0, duration:(dur==null?0.8:dur), ease:"power2.inOut"}, at||0);
}
function countUp(tl, sel, from, to, at, dur, decimals){ // counts a number in #id text
  var o = {v:from};
  tl.to(o, {v:to, duration:(dur==null?1.2:dur), ease:"power2.out",
    onUpdate:function(){ var el=document.querySelector(sel); if(el) el.textContent=o.v.toFixed(decimals||0); }}, at||0);
}
</script>
